<script>
  // PASTE YOUR GOOGLE APPS SCRIPT WEB APP URL HERE
  const SCRIPT_URL = "https://script.google.com/macros/s/AKfycbyMcl86dH2DaOtcOlr9jsUOKxb3jtzFGuuc40U4POB7b_K0KBAU8jcK0ws0iMNNFva8/exec";

  let database = [];
  let selectedParty = null;
  let responses = {};

  // Live Countdown (Target: November 27, 2026)
  const weddingDate = new Date("November 27, 2026 15:00:00").getTime();

  function updateCountdown() {
    const now = new Date().getTime();
    const distance = weddingDate - now;

    if (distance > 0) {
      const days = Math.floor(distance / (1000 * 60 * 60 * 24));
      const hours = Math.floor((distance % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60));
      const minutes = Math.floor((distance % (1000 * 60 * 60)) / (1000 * 60));
      const seconds = Math.floor((distance % (1000 * 60)) / 1000);

      document.getElementById("cd-days").textContent = String(days).padStart(2, '0');
      document.getElementById("cd-hours").textContent = String(hours).padStart(2, '0');
      document.getElementById("cd-mins").textContent = String(minutes).padStart(2, '0');
      document.getElementById("cd-secs").textContent = String(seconds).padStart(2, '0');
    }
  }

  setInterval(updateCountdown, 1000);
  updateCountdown();

  function triggerNoticeModal() {
    document.getElementById('notice-modal').style.display = 'flex';
  }

  function closeModal() {
    document.getElementById('notice-modal').style.display = 'none';
  }

  window.addEventListener('DOMContentLoaded', () => {
    if (SCRIPT_URL !== "YOUR_APPS_SCRIPT_WEB_APP_URL_HERE") {
      fetch(`${SCRIPT_URL}?action=getGuestList`)
        .then(res => res.json())
        .then(data => { database = data; })
        .catch(err => console.error("Error loading guest data:", err));
    }
  });

  const searchInput = document.getElementById('search-input');
  const dropdown = document.getElementById('dropdown');

  searchInput.addEventListener('input', () => {
    const query = searchInput.value.toLowerCase().trim();
    dropdown.innerHTML = '';
    
    if (!query) {
      dropdown.style.display = 'none';
      return;
    }

    const matches = database.filter(p => 
      p.partyTitle.toLowerCase().includes(query) || 
      p.members.some(m => m.toLowerCase().includes(query))
    );

    if (matches.length > 0) {
      dropdown.style.display = 'block';
      matches.forEach(p => {
        const item = document.createElement('div');
        item.textContent = p.partyTitle;
        item.onclick = () => {
          searchInput.value = p.partyTitle;
          selectedParty = p;
          dropdown.style.display = 'none';
        };
        dropdown.appendChild(item);
      });
    } else {
      dropdown.style.display = 'none';
    }
  });

  function findInvitation() {
    if (!selectedParty) {
      const query = searchInput.value.toLowerCase().trim();
      selectedParty = database.find(p => 
        p.partyTitle.toLowerCase().includes(query) || 
        p.members.some(m => m.toLowerCase().includes(query))
      );
    }

    if (!selectedParty) {
      alert("Name not found in guest list. Please check spelling or type the main invitee name.");
      return;
    }

    // Returning guest check
    if (selectedParty.hasResponded) {
      renderAlreadySubmitted();
      showStep('step-already-submitted');
      return;
    }

    renderForm();
    showStep('step-form');
  }

  function renderAlreadySubmitted() {
    const listEl = document.getElementById('already-submitted-list');
    listEl.innerHTML = '';

    selectedParty.members.forEach(member => {
      const item = document.createElement('div');
      item.style.cssText = 'display:flex; justify-content:space-between; margin-bottom:8px;';
      item.innerHTML = `
        <span style="font-size: 13.5px;">${member}</span>
        <span class="badge badge-attending">Submitted</span>
      `;
      listEl.appendChild(item);
    });
  }

  function renderForm() {
    document.getElementById('party-title').textContent = selectedParty.partyTitle;
    document.getElementById('seat-count').textContent = selectedParty.reservedSeats;

    const guestListEl = document.getElementById('guest-list');
    guestListEl.innerHTML = '';
    responses = {};

    selectedParty.members.forEach((guest) => {
      const card = document.createElement('div');
      card.className = 'guest-option';
      card.innerHTML = `
        <div class="guest-name">${guest}</div>
        <div class="radio-group">
          <div class="radio-btn" id="attend-${guest}" onclick="setAttendance('${guest}', true)">Attending</div>
          <div class="radio-btn" id="decline-${guest}" onclick="setAttendance('${guest}', false)">Not Attending</div>
        </div>
      `;
      guestListEl.appendChild(card);
    });

    updateProgress();
  }

  function setAttendance(guest, isAttending) {
    responses[guest] = isAttending;
    document.getElementById(`attend-${guest}`).className = isAttending ? 'radio-btn selected-attending' : 'radio-btn';
    document.getElementById(`decline-${guest}`).className = !isAttending ? 'radio-btn selected-declining' : 'radio-btn';
    updateProgress();
  }

  function updateProgress() {
    const answeredCount = Object.keys(responses).length;
    const totalCount = selectedParty.members.length;
    document.getElementById('progress-tracker').textContent = `${answeredCount} of ${totalCount} answered`;
  }

  function goToReview() {
    if (Object.keys(responses).length < selectedParty.members.length) {
      alert("Please select an RSVP response for every guest.");
      return;
    }

    const attendingCount = Object.values(responses).filter(v => v === true).length;
    document.getElementById('review-summary-text').textContent = `${attendingCount} of ${selectedParty.members.length} attending`;

    const reviewList = document.getElementById('review-list');
    reviewList.innerHTML = '';

    for (const [guest, isAttending] of Object.entries(responses)) {
      const item = document.createElement('div');
      item.style.cssText = 'display:flex; justify-content:space-between; margin-bottom:8px;';
      item.innerHTML = `
        <span style="font-size: 13.5px;">${guest}</span>
        <span class="badge ${isAttending ? 'badge-attending' : 'badge-declining'}">
          ${isAttending ? 'Attending' : 'Not Attending'}
        </span>
      `;
      reviewList.appendChild(item);
    }

    showStep('step-review');
  }

  function backToForm() { showStep('step-form'); }

  function submitRSVP() {
    const attendingNames = Object.entries(responses)
      .filter(([_, isAttending]) => isAttending)
      .map(([name]) => name)
      .join(', ');

    const decliningNames = Object.entries(responses)
      .filter(([_, isAttending]) => !isAttending)
      .map(([name]) => name)
      .join(', ');

    const attendingCount = Object.values(responses).filter(v => v === true).length;
    const decliningCount = Object.values(responses).filter(v => v === false).length;

    let overallStatus = "Declined";
    if (attendingCount === selectedParty.members.length) {
      overallStatus = "Attending";
    } else if (attendingCount > 0) {
      overallStatus = "Partially Attending";
    }

    const formData = new URLSearchParams();
    formData.append('row', selectedParty.row);
    formData.append('attendanceStatus', overallStatus);
    formData.append('phone', document.getElementById('phone').value || '');
    formData.append('attendingNames', attendingNames);
    formData.append('decliningNames', decliningNames);
    formData.append('attendingCount', attendingCount);
    formData.append('decliningCount', decliningCount);

    // Instant UI transition
    document.getElementById('final-list').innerHTML = document.getElementById('review-list').innerHTML;
    showStep('step-confirmed');

    // Silent background save
    fetch(SCRIPT_URL, {
      method: 'POST',
      mode: 'no-cors',
      headers: {
        'Content-Type': 'application/x-www-form-urlencoded',
      },
      body: formData.toString()
    }).catch(err => {
      console.error("Background sync error:", err);
    });
  }

  function resetForm() {
    selectedParty = null;
    responses = {};
    searchInput.value = '';
    showStep('step-search');
  }

  function showStep(stepId) {
    ['step-search', 'step-form', 'step-review', 'step-already-submitted', 'step-confirmed'].forEach(id => {
      document.getElementById(id).classList.add('hidden');
    });
    document.getElementById(stepId).classList.remove('hidden');
  }
</script>
